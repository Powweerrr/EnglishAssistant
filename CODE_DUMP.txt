from __future__ import annotations

import sys

from PySide6.QtCore import QObject, QThread, QTimer, Signal
from PySide6.QtGui import QCursor, QPixmap
from PySide6.QtWidgets import QApplication

from english_assistant.capture.screen_capture import capture_region
from english_assistant.capture.screen_selector import ScreenSelector
from english_assistant.lookup.worker import QuickLookupWorker
from english_assistant.ocr.worker import OcrWorker
from english_assistant.ui.lookup_overlay import LookupOverlay
from english_assistant.ui.main_window import MainWindow


class HotkeyBridge(QObject):
    capture_triggered = Signal()
    quick_lookup_triggered = Signal()


def main() -> int:
    app = QApplication(sys.argv)
    app.setApplicationName("EnglishAssistant")
    app.setOrganizationName("EnglishAssistant")
    window = MainWindow()
    selector = ScreenSelector()
    overlay = LookupOverlay()
    workers: list[QThread] = []
    bridge = HotkeyBridge()
    lookup_generation = 0

    def track_worker(worker):
        workers.append(worker)

        def finished():
            if worker in workers:
                workers.remove(worker)
            worker.deleteLater()

        worker.finished.connect(finished)
        worker.start()

    def start_ocr(pixmap: QPixmap):
        window.set_ocr_busy(True)
        worker = OcrWorker(pixmap, window)
        worker.succeeded.connect(window.set_ocr_text)
        worker.failed.connect(window.show_error)

        def finished():
            window.set_ocr_busy(False)

        worker.finished.connect(finished)
        track_worker(worker)

    def start_capture():
        overlay.hide()
        window.hide()
        QTimer.singleShot(180, selector.show_selector)

    def selected(region):
        def process():
            try:
                pixmap = capture_region(region)
                window.set_preview(pixmap)
                window.show()
                window.raise_()
                window.activateWindow()
                start_ocr(pixmap)
            except (RuntimeError, ValueError, OSError) as error:
                window.show()
                window.show_error(str(error))
        QTimer.singleShot(160, process)

    def cancelled():
        window.show()
        window.raise_()
        window.activateWindow()

    def start_quick_lookup():
        nonlocal lookup_generation
        lookup_generation += 1
        generation = lookup_generation
        if overlay.isVisible():
            overlay.hide()
            return
        point = QCursor.pos()
        overlay.show_loading()
        worker = QuickLookupWorker(point.x(), point.y(), window)
        worker.succeeded.connect(
            lambda result, request_id=generation:
                overlay.show_result(result) if request_id == lookup_generation else None
        )
        worker.failed.connect(
            lambda message, request_id=generation:
                overlay.show_error(message) if request_id == lookup_generation else None
        )
        track_worker(worker)

    window.capture_requested.connect(start_capture)
    window.quick_lookup_requested.connect(start_quick_lookup)
    bridge.capture_triggered.connect(start_capture)
    bridge.quick_lookup_triggered.connect(start_quick_lookup)
    selector.region_selected.connect(selected)
    selector.selection_cancelled.connect(cancelled)

    hotkey_listener = None
    try:
        from pynput import keyboard
        hotkey_listener = keyboard.GlobalHotKeys({
            "<ctrl>+<shift>+e": bridge.capture_triggered.emit,
            "<alt>+q": bridge.quick_lookup_triggered.emit,
        })
        hotkey_listener.start()
    except (ImportError, OSError):
        window.status_label.setText("Готов · глобальные клавиши недоступны в этой системе")

    def wait_workers():
        if hotkey_listener is not None:
            hotkey_listener.stop()
        for worker in tuple(workers):
            worker.wait(5000)

    app.aboutToQuit.connect(wait_workers)
    window.show()
    return app.exec()


if __name__ == "__main__":
    raise SystemExit(main())

from __future__ import annotations

from dataclasses import dataclass

from PIL import Image


@dataclass(frozen=True, slots=True)
class CursorCapture:
    image: Image.Image
    origin_x: int
    origin_y: int
    cursor_x: int
    cursor_y: int


def capture_around_cursor(x: int, y: int, width: int = 520, height: int = 220) -> CursorCapture:
    """Capture a small physical-pixel region around the pointer using Desktop Duplication."""
    try:
        import mss
    except ImportError as error:  # pragma: no cover - dependency is installed by the app
        raise RuntimeError("Для поиска слова под курсором не установлен пакет mss.") from error

    with mss.mss() as capture:
        monitors = capture.monitors[1:]
        monitor = next(
            (m for m in monitors
             if m["left"] <= x < m["left"] + m["width"]
             and m["top"] <= y < m["top"] + m["height"]),
            None,
        )
        if monitor is None:
            raise RuntimeError("Не удалось определить монитор под курсором.")

        left = max(monitor["left"], min(x - width // 2, monitor["left"] + monitor["width"] - width))
        top = max(monitor["top"], min(y - height // 2, monitor["top"] + monitor["height"] - height))
        right = min(monitor["left"] + monitor["width"], left + width)
        bottom = min(monitor["top"] + monitor["height"], top + height)
        region = {"left": left, "top": top, "width": right - left, "height": bottom - top}
        shot = capture.grab(region)
        image = Image.frombytes("RGB", shot.size, shot.rgb)
        return CursorCapture(image, left, top, x, y)

"""Full-screen screen capture overlay for EnglishAssistant."""

from __future__ import annotations

import mss
from PIL import Image
from PySide6.QtCore import QPoint, QRect, Qt, Signal
from PySide6.QtGui import (
    QColor,
    QGuiApplication,
    QImage,
    QKeyEvent,
    QMouseEvent,
    QPainter,
    QPaintEvent,
    QPen,
    QPixmap,
)
from PySide6.QtWidgets import QWidget


class ScreenCaptureOverlay(QWidget):
    """Frameless top-most widget that lets the user select a screen region."""

    image_selected = Signal(object)
    canceled = Signal()

    MIN_SELECTION_SIZE = 5
    MASK_ALPHA = 120

    def __init__(self, parent: QWidget | None = None) -> None:
        super().__init__(parent)

        self.setWindowFlags(
            Qt.WindowType.FramelessWindowHint
            | Qt.WindowType.WindowStaysOnTopHint
            | Qt.WindowType.Tool
        )
        # Окно непрозрачное, мы сами рисуем замороженный скриншот.
        # WA_TranslucentBackground вызывал черный экран на Windows.
        self.setAttribute(Qt.WidgetAttribute.WA_OpaquePaintEvent, True)
        self.setAttribute(Qt.WidgetAttribute.WA_DeleteOnClose, False)
        self.setFocusPolicy(Qt.FocusPolicy.StrongFocus)
        self.setMouseTracking(True)
        self.setCursor(Qt.CursorShape.CrossCursor)

        self._full_image: Image.Image | None = None
        self._background_pixmap = QPixmap()
        self._origin: QPoint | None = None
        self._current: QPoint | None = None
        self._dragging = False

    def start_capture(self) -> None:
        if self.isVisible():
            return

        geometry = self._compute_virtual_geometry()
        self.setGeometry(geometry)

        try:
            self._full_image = self._grab_all_screens()
        except Exception:
            self._full_image = None
            self._background_pixmap = QPixmap()
            self.hide()
            self.canceled.emit()
            return

        self._background_pixmap = self._pil_to_pixmap(self._full_image).scaled(
            self.size(),
            Qt.AspectRatioMode.IgnoreAspectRatio,
            Qt.TransformationMode.SmoothTransformation,
        )

        self._reset_selection()
        self.show()
        self.raise_()
        self.activateWindow()
        self.setFocus(Qt.FocusReason.ActiveWindowFocusReason)

    def cancel_capture(self) -> None:
        self.hide()
        self._reset_selection()
        self._full_image = None
        self._background_pixmap = QPixmap()
        self.canceled.emit()

    def _compute_virtual_geometry(self) -> QRect:
        screens = QGuiApplication.screens()
        if not screens:
            return QRect(0, 0, 1, 1)

        geometry = QRect(screens[0].geometry())
        for screen in screens[1:]:
            geometry = geometry.united(screen.geometry())
        return geometry

    def _grab_all_screens(self) -> Image.Image:
        with mss.mss() as sct:
            monitor = sct.monitors[0]
            shot = sct.grab(monitor)
            return Image.frombytes(
                "RGB",
                shot.size,
                shot.bgra,
                "raw",
                "BGRX",
            )

    def _pil_to_pixmap(self, image: Image.Image) -> QPixmap:
        rgba = image.convert("RGBA")
        data = rgba.tobytes("raw", "RGBA")
        qimage = QImage(
            data,
            rgba.width,
            rgba.height,
            rgba.width * 4,
            QImage.Format.Format_RGBA8888,
        ).copy()
        return QPixmap.fromImage(qimage)

    def _reset_selection(self) -> None:
        self._origin = None
        self._current = None
        self._dragging = False
        self.update()

    def _selection_rect(self) -> QRect:
        if self._origin is None or self._current is None:
            return QRect()
        rect = QRect(self._origin, self._current).normalized()
        return rect.intersected(self.rect())

    def _crop_selected_image(self, rect: QRect) -> Image.Image:
        if self._full_image is None:
            raise RuntimeError("No screenshot is available for cropping.")

        image_width, image_height = self._full_image.size
        widget_width = max(1, self.width())
        widget_height = max(1, self.height())

        scale_x = image_width / widget_width
        scale_y = image_height / widget_height

        left = int(round(rect.left() * scale_x))
        top = int(round(rect.top() * scale_y))
        right = int(round((rect.left() + rect.width()) * scale_x))
        bottom = int(round((rect.top() + rect.height()) * scale_y))

        left = max(0, min(left, image_width - 1))
        top = max(0, min(top, image_height - 1))
        right = max(left + 1, min(right, image_width))
        bottom = max(top + 1, min(bottom, image_height))

        return self._full_image.crop((left, top, right, bottom))

    def _finish_selection(self) -> None:
        rect = self._selection_rect()

        if (
            rect.width() < self.MIN_SELECTION_SIZE
            or rect.height() < self.MIN_SELECTION_SIZE
        ):
            self.cancel_capture()
            return

        try:
            image = self._crop_selected_image(rect)
        except Exception:
            self.cancel_capture()
            return

        self.hide()
        self._reset_selection()
        self._full_image = None
        self._background_pixmap = QPixmap()
        self.image_selected.emit(image)

    def mousePressEvent(self, event: QMouseEvent) -> None:
        if event.button() == Qt.MouseButton.LeftButton:
            point = event.position().toPoint()
            self._origin = point
            self._current = point
            self._dragging = True
            self.update()
            event.accept()
            return
        super().mousePressEvent(event)

    def mouseMoveEvent(self, event: QMouseEvent) -> None:
        if self._dragging:
            self._current = event.position().toPoint()
            self.update()
            event.accept()
            return
        super().mouseMoveEvent(event)

    def mouseReleaseEvent(self, event: QMouseEvent) -> None:
        if event.button() == Qt.MouseButton.LeftButton and self._dragging:
            self._current = event.position().toPoint()
            self._dragging = False
            self._finish_selection()
            event.accept()
            return
        super().mouseReleaseEvent(event)

    def keyPressEvent(self, event: QKeyEvent) -> None:
        if event.key() == Qt.Key.Key_Escape:
            self.cancel_capture()
            event.accept()
            return
        super().keyPressEvent(event)

    def paintEvent(self, event: QPaintEvent) -> None:
        painter = QPainter(self)
        painter.setRenderHint(QPainter.RenderHint.SmoothPixmapTransform, True)

        # 1. Фон (замороженный скриншот)
        if not self._background_pixmap.isNull():
            painter.drawPixmap(self.rect(), self._background_pixmap)
        else:
            painter.fillRect(self.rect(), QColor(30, 30, 30))

        # 2. Затемнение
        painter.fillRect(self.rect(), QColor(0, 0, 0, self.MASK_ALPHA))

        # 3. "Дыра" (чистый скриншот внутри рамки)
        selection = self._selection_rect()
        if not selection.isEmpty() and not self._background_pixmap.isNull():
            painter.save()
            painter.setClipRect(selection)
            painter.drawPixmap(self.rect(), self._background_pixmap)
            painter.restore()

            # 4. Рамка и текст
            painter.setPen(QPen(QColor(0, 180, 255), 2))
            painter.setBrush(Qt.BrushStyle.NoBrush)
            painter.drawRect(selection.adjusted(0, 0, -1, -1))

            painter.setPen(QColor(255, 255, 255))
            painter.drawText(
                selection.adjusted(4, 4, 0, 0),
                Qt.AlignmentFlag.AlignLeft | Qt.AlignmentFlag.AlignTop,
                f"{selection.width()} x {selection.height()}",
            )

        painter.end()

from PySide6.QtCore import QRect
from PySide6.QtGui import QPixmap
from PySide6.QtWidgets import QApplication


def capture_region(region: QRect) -> QPixmap:
    screen = QApplication.primaryScreen()
    if screen is None:
        raise RuntimeError("Не удалось определить основной монитор.")
    if region.isEmpty():
        raise ValueError("Выделенная область пуста.")
    pixmap = screen.grabWindow(0, region.x(), region.y(), region.width(), region.height())
    if pixmap.isNull():
        raise RuntimeError("Не удалось получить снимок экрана.")
    return pixmap

from PySide6.QtCore import QPoint, QRect, Qt, Signal
from PySide6.QtGui import QColor, QPainter, QPen
from PySide6.QtWidgets import QApplication, QWidget


class ScreenSelector(QWidget):
    region_selected = Signal(QRect)
    selection_cancelled = Signal()

    def __init__(self):
        super().__init__()
        self.setWindowFlags(Qt.WindowType.FramelessWindowHint | Qt.WindowType.WindowStaysOnTopHint | Qt.WindowType.Tool)
        self.setAttribute(Qt.WidgetAttribute.WA_TranslucentBackground)
        self.setMouseTracking(True)
        self.setFocusPolicy(Qt.FocusPolicy.StrongFocus)
        self.setCursor(Qt.CursorShape.CrossCursor)
        self.start_point = QPoint()
        self.current_point = QPoint()
        self.selecting = False

    def show_selector(self):
        screen = QApplication.primaryScreen()
        if screen is None:
            self.selection_cancelled.emit()
            return
        self.start_point = QPoint()
        self.current_point = QPoint()
        self.selecting = False
        self.setGeometry(screen.geometry())
        self.show()
        self.raise_()
        self.activateWindow()
        self.setFocus()

    def selection_rect(self) -> QRect:
        return QRect(self.start_point, self.current_point).normalized()

    def paintEvent(self, event):
        del event
        painter = QPainter(self)
        painter.fillRect(self.rect(), QColor(0, 0, 0, 115))
        if self.selecting:
            rect = self.selection_rect()
            painter.setCompositionMode(QPainter.CompositionMode.CompositionMode_Clear)
            painter.fillRect(rect, Qt.GlobalColor.transparent)
            painter.setCompositionMode(QPainter.CompositionMode.CompositionMode_SourceOver)
            painter.setPen(QPen(QColor(70, 170, 255), 2))
            painter.drawRect(rect)

    def mousePressEvent(self, event):
        if event.button() == Qt.MouseButton.LeftButton:
            self.start_point = event.position().toPoint()
            self.current_point = self.start_point
            self.selecting = True
            self.update()

    def mouseMoveEvent(self, event):
        if self.selecting:
            self.current_point = event.position().toPoint()
            self.update()

    def mouseReleaseEvent(self, event):
        if event.button() != Qt.MouseButton.LeftButton or not self.selecting:
            return
        self.current_point = event.position().toPoint()
        self.selecting = False
        rect = self.selection_rect()
        self.hide()
        if rect.width() < 5 or rect.height() < 5:
            self.selection_cancelled.emit()
        else:
            self.region_selected.emit(rect)

    def keyPressEvent(self, event):
        if event.key() == Qt.Key.Key_Escape:
            self.selecting = False
            self.hide()
            self.selection_cancelled.emit()
            return
        super().keyPressEvent(event)

from __future__ import annotations

import gzip
from functools import lru_cache
import html
import re
import struct
import urllib.request
import zipfile
from dataclasses import dataclass
from pathlib import Path

DEFAULT_DOWNLOAD_URL = (
    "https://github.com/DanielGregorini/koreader-dicts/releases/latest/download/en-ru.zip"
)


@dataclass(frozen=True, slots=True)
class DictionaryEntry:
    headword: str
    article: str


class StarDictError(RuntimeError):
    pass


def _plain_article(raw: bytes) -> str:
    text = raw.decode("utf-8", errors="replace")
    text = re.sub(r"(?i)<br\s*/?>", "\n", text)
    text = re.sub(r"(?i)</(p|div|li)>", "\n", text)
    text = re.sub(r"<[^>]+>", "", text)
    text = html.unescape(text)
    return "\n".join(line.strip() for line in text.splitlines() if line.strip()).strip()


class StarDictReader:
    """Small read-only StarDict reader for .ifo/.idx/.dict.dz/.syn dictionaries."""

    def __init__(self, directory: str | Path):
        self.directory = Path(directory)
        ifo_files = list(self.directory.glob("*.ifo"))
        if len(ifo_files) != 1:
            raise StarDictError(f"Ожидался один .ifo файл в {self.directory}")
        self.prefix = ifo_files[0].with_suffix("")
        self._entries: dict[str, tuple[int, int]] = {}
        self._synonyms: dict[str, str] = {}
        self._dict_data: bytes | None = None
        self._load_index()

    def _load_index(self) -> None:
        idx_path = self.prefix.with_suffix(".idx")
        dict_path = self.prefix.with_suffix(".dict.dz")
        if not dict_path.exists():
            dict_path = self.prefix.with_suffix(".dict")
        if not idx_path.exists() or not dict_path.exists():
            raise StarDictError("В словаре отсутствует .idx или .dict/.dict.dz файл.")

        data = idx_path.read_bytes()
        cursor = 0
        while cursor < len(data):
            end = data.find(b"\0", cursor)
            if end < 0 or end + 9 > len(data):
                raise StarDictError("Повреждён индекс StarDict (.idx).")
            word = data[cursor:end].decode("utf-8", errors="replace").casefold()
            offset, size = struct.unpack_from(">II", data, end + 1)
            self._entries[word] = (offset, size)
            cursor = end + 9

        headwords = list(self._entries)
        syn_path = self.prefix.with_suffix(".syn")
        if syn_path.exists():
            syn_data = syn_path.read_bytes()
            cursor = 0
            while cursor < len(syn_data):
                end = syn_data.find(b"\0", cursor)
                if end < 0 or end + 5 > len(syn_data):
                    raise StarDictError("Повреждён индекс словоформ (.syn).")
                form = syn_data[cursor:end].decode("utf-8", errors="replace").casefold()
                index = struct.unpack_from(">I", syn_data, end + 1)[0]
                if index < len(headwords):
                    # StarDict .syn references the sorted .idx record number.
                    self._synonyms[form] = headwords[index]
                cursor = end + 5
        self._dict_path = dict_path

    def _data(self) -> bytes:
        if self._dict_data is None:
            raw = self._dict_path.read_bytes()
            try:
                self._dict_data = gzip.decompress(raw) if self._dict_path.suffix == ".dz" else raw
            except OSError as error:
                raise StarDictError("Не удалось распаковать словарную базу.") from error
        return self._dict_data

    def lookup(self, word: str) -> DictionaryEntry | None:
        key = word.strip().casefold()
        if not key:
            return None
        headword = key if key in self._entries else self._synonyms.get(key)
        if headword is None:
            return None
        offset, size = self._entries[headword]
        article = _plain_article(self._data()[offset:offset + size])
        if not article:
            return None
        return DictionaryEntry(headword=headword, article=article)


def install_en_ru_dictionary(destination: str | Path, *, url: str = DEFAULT_DOWNLOAD_URL) -> Path:
    """Download and safely install the CC BY-SA 4.0 en-ru StarDict archive."""
    destination = Path(destination)
    destination.mkdir(parents=True, exist_ok=True)
    archive_path = destination / "en-ru.download.zip"
    request = urllib.request.Request(url, headers={"User-Agent": "EnglishAssistant/1.2"})
    try:
        with urllib.request.urlopen(request, timeout=45) as response, archive_path.open("wb") as out:
            while chunk := response.read(1024 * 1024):
                out.write(chunk)
        with zipfile.ZipFile(archive_path) as archive:
            root = destination.resolve()
            for member in archive.infolist():
                target = (destination / member.filename).resolve()
                if root not in target.parents and target != root:
                    raise StarDictError("Архив содержит недопустимый путь.")
            archive.extractall(destination)
    except Exception:
        archive_path.unlink(missing_ok=True)
        raise
    finally:
        archive_path.unlink(missing_ok=True)

    candidates = list(destination.rglob("*.ifo"))
    if len(candidates) != 1:
        raise StarDictError("Архив загружен, но структура словаря неожиданная.")
    StarDictReader(candidates[0].parent)  # validate before reporting success
    get_default_dictionary.cache_clear()
    return candidates[0].parent


@lru_cache(maxsize=1)
def get_default_dictionary() -> StarDictReader | None:
    root = Path.home() / ".english_assistant" / "dictionaries" / "en-ru"
    if not root.exists():
        return None
    ifo_files = list(root.rglob("*.ifo"))
    if len(ifo_files) != 1:
        return None
    try:
        return StarDictReader(ifo_files[0].parent)
    except (OSError, ValueError, StarDictError):
        return None

class DomainError(RuntimeError):
    """An actionable error from a language domain."""


@dataclass(frozen=True, slots=True)
class DomainResult:
    domain: str
    title: str
    source_text: str
    content: str
    metadata: dict


def request_json(url: str, *, params: dict | None = None, data: dict | None = None,
                 timeout: float = 18.0) -> dict | list:
    if params:
        url = f"{url}?{urlencode(params)}"
    body = urlencode(data).encode("utf-8") if data is not None else None
    headers = {
        "User-Agent": "EnglishAssistant/1.0 (desktop language learning tool)",
        "Accept": "application/json",
    }
    if body is not None:
        headers["Content-Type"] = "application/x-www-form-urlencoded; charset=UTF-8"
    request = Request(url, data=body, headers=headers,
                      method="POST" if body is not None else "GET")
    try:
        with urlopen(request, timeout=timeout) as response:
            return json.loads(response.read().decode("utf-8"))
    except HTTPError as exc:
        detail = exc.read(400).decode("utf-8", errors="replace")
        raise DomainError(f"Сервис вернул HTTP {exc.code}: {detail[:250]}") from exc
    except (URLError, TimeoutError, OSError) as exc:
        raise DomainError(f"Не удалось подключиться к сервису: {exc}") from exc
    except (UnicodeDecodeError, json.JSONDecodeError) as exc:
        raise DomainError("Сервис вернул некорректный ответ.") from exc


class TranslationDomain:
    endpoint = "https://api.mymemory.translated.net/get"

    def __init__(self, timeout: float = 18.0):
        self.timeout = timeout

    def run(self, text: str, source: str = "en", target: str = "ru") -> DomainResult:
        text = text.strip()
        if not text:
            raise DomainError("Введите текст для перевода.")
        if len(text) > 4500:
            raise DomainError("Максимальная длина одного запроса — 4500 символов.")
        if source == target:
            raise DomainError("Исходный и целевой языки должны различаться.")
        payload = request_json(self.endpoint,
                               params={"q": text, "langpair": f"{source}|{target}"}, timeout=self.timeout)
        if not isinstance(payload, dict) or payload.get("responseStatus") != 200:
            detail = payload.get("responseDetails", "Перевод не выполнен.") if isinstance(payload, dict) else "Некорректный ответ сервиса."
            raise DomainError(str(detail))
        data = payload.get("responseData") or {}
        translated = data.get("translatedText")
        if not isinstance(translated, str) or not translated.strip():
            raise DomainError("Сервис не вернул перевод.")
        return DomainResult("translation", "Перевод", text, unescape(translated).strip(),
                            {"source": source, "target": target,
                             "match": data.get("match")})


class DictionaryDomain:
    def __init__(self, timeout: float = 18.0):
        self.timeout = timeout

    endpoint = "https://api.dictionaryapi.dev/api/v2/entries/en/{}"

    def run(self, term: str) -> DomainResult:
        term = term.strip()
        if not term or len(term) > 100:
            raise DomainError("Введите английское слово или выражение длиной до 100 символов.")
        try:
            from english_assistant.dictionary import get_default_dictionary
            local = get_default_dictionary()
            entry = local.lookup(term) if local is not None else None
        except Exception:
            entry = None
        if entry is not None:
            return DomainResult(
                "dictionary", f"Локальный словарь: {entry.headword}", term,
                entry.article, {"word": term, "headword": entry.headword, "source": "offline-en-ru"},
            )
        payload = request_json(self.endpoint.format(quote(term)), timeout=self.timeout)
        if not isinstance(payload, list) or not payload:
            raise DomainError(f"Словарная статья для «{term}» не найдена.")
        entry = payload[0]
        lines: list[str] = []
        phonetics = list(dict.fromkeys(p.get("text") for p in entry.get("phonetics", []) if p.get("text")))
        if phonetics:
            lines.append("Произношение: " + " · ".join(phonetics))
        for meaning in entry.get("meanings", []):
            lines.append(f"\n{meaning.get('partOfSpeech', 'Часть речи').capitalize()}")
            for number, item in enumerate(meaning.get("definitions", [])[:8], 1):
                definition = item.get("definition", "")
                if definition:
                    lines.append(f"{number}. {definition}")
                if item.get("example"):
                    lines.append(f"   Пример: {item['example']}")
                if item.get("synonyms"):
                    lines.append("   Синонимы: " + ", ".join(item["synonyms"][:8]))
        content = "\n".join(lines).strip()
        if not content:
            raise DomainError("Словарная статья не содержит определений.")
        return DomainResult("dictionary", f"Словарь: {term}", term, content,
                            {"word": term, "phonetics": phonetics})


class GrammarDomain:
    endpoint = "https://api.languagetool.org/v2/check"

    def run(self, text: str, language: str = "en-US") -> DomainResult:
        text = text.strip()
        if not text:
            raise DomainError("Введите предложение для проверки.")
        if len(text) > 12000:
            raise DomainError("Максимальная длина текста — 12000 символов.")
        payload = request_json(self.endpoint, data={"text": text, "language": language}, timeout=30)
        matches = payload.get("matches", []) if isinstance(payload, dict) else []
        if not matches:
            content = "Замечаний не найдено. Автоматическая проверка не гарантирует абсолютную правильность текста."
        else:
            blocks = [f"Найдено замечаний: {len(matches)}", ""]
            for number, item in enumerate(matches[:40], 1):
                start, length = int(item.get("offset", 0)), int(item.get("length", 0))
                fragment = text[start:start + length]
                blocks.append(f"{number}. {item.get('message', 'Замечание')}")
                if fragment:
                    blocks.append(f"   Фрагмент: «{fragment}»")
                replacements = [r.get("value") for r in item.get("replacements", [])[:6] if r.get("value")]
                if replacements:
                    blocks.append("   Возможные варианты: " + ", ".join(replacements))
                category = (item.get("rule") or {}).get("category", {}).get("name")
                if category:
                    blocks.append(f"   Категория: {category}")
                blocks.append("")
            content = "\n".join(blocks).strip()
        return DomainResult("grammar", "Грамматика", text, content,
                            {"issue_count": len(matches), "language": language})

        else:
            content = "Совпадений во встроенном справочнике не найдено. Список выражений не является исчерпывающим."
        return DomainResult("expressions", "Идиомы и фразовые глаголы", text, content,
                            {"count": len(found), "matches": [x[0] for x in found]})

class CefrDomain:
    def run(self, text: str) -> DomainResult:
        text = text.strip()
        if not text:
            raise DomainError("Введите текст для анализа лексики.")
        words = re.findall(r"[a-z]+(?:['’-][a-z]+)?", text.casefold())
        if not words:
            raise DomainError("Английские слова не найдены.")
        matches = [(word, CEFR_WORDS[word]) for word in words if word in CEFR_WORDS]
        counts: dict[str, int] = {}
        for _, level in matches:
            counts[level] = counts.get(level, 0) + 1
        if not matches:
            content = "Слова не найдены во встроенном справочнике. Надёжно оценить CEFR по этому тексту нельзя."
        else:
            rows = ["Ориентировочная оценка известных слов:"]
            rows.extend(f"• {word} — {level}" for word, level in matches)
            rows.append("\nРаспределение: " + ", ".join(f"{k}: {v}" for k, v in sorted(counts.items())))
            rows.append("\nЭто ограниченный справочник отдельных слов, а не официальный тест уровня владения языком.")
            content = "\n".join(rows)
        return DomainResult("cefr", "Лексика и CEFR", text, content,
                            {"total_words": len(words), "matched_words": len(matches), "distribution": counts})


class VocabularyStore:
    def __init__(self, path: str | Path | None = None):
        base = Path.home() / ".english_assistant"
        self.path = Path(path) if path else base / "english_assistant.sqlite3"
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self._init_db()

    def connect(self) -> sqlite3.Connection:
        db = sqlite3.connect(self.path, timeout=15)
        db.row_factory = sqlite3.Row
        return db

    def _init_db(self) -> None:
        with self.connect() as db:
            db.execute("""CREATE TABLE IF NOT EXISTS vocabulary (
                id INTEGER PRIMARY KEY AUTOINCREMENT, term TEXT NOT NULL COLLATE NOCASE,
                meaning TEXT NOT NULL DEFAULT '', example TEXT NOT NULL DEFAULT '',
                source_text TEXT NOT NULL DEFAULT '', created_at TEXT NOT NULL,
                UNIQUE(term, meaning))""")
            db.execute("""CREATE TABLE IF NOT EXISTS history (
                id INTEGER PRIMARY KEY AUTOINCREMENT, domain TEXT NOT NULL,
                source_text TEXT NOT NULL, result_text TEXT NOT NULL, created_at TEXT NOT NULL)""")
            db.execute("""CREATE TABLE IF NOT EXISTS quick_lookup_cache (
                term TEXT PRIMARY KEY COLLATE NOCASE,
                translation TEXT NOT NULL DEFAULT '',
                definition TEXT NOT NULL DEFAULT '',
                phonetics TEXT NOT NULL DEFAULT '',
                updated_at TEXT NOT NULL)""")

    @staticmethod
    def now() -> str:
        return datetime.now(timezone.utc).isoformat(timespec="seconds")

    def add_word(self, term: str, meaning: str = "", example: str = "", source_text: str = "") -> int:
        term = term.strip()
        if not term:
            raise ValueError("Нельзя сохранить пустое слово.")
        with self.connect() as db:
            db.execute("INSERT OR IGNORE INTO vocabulary(term,meaning,example,source_text,created_at) VALUES(?,?,?,?,?)",
                       (term, meaning.strip(), example.strip(), source_text.strip(), self.now()))
            row = db.execute("SELECT id FROM vocabulary WHERE term=? AND meaning=?",
                             (term, meaning.strip())).fetchone()
            return int(row["id"])

    def delete_word(self, word_id: int) -> bool:
        with self.connect() as db:
            cur = db.execute("DELETE FROM vocabulary WHERE id=?", (word_id,))
            return cur.rowcount > 0

    def words(self) -> list[dict]:
        with self.connect() as db:
            return [dict(r) for r in db.execute("SELECT * FROM vocabulary ORDER BY id DESC")]

    def add_history(self, result: DomainResult) -> None:
        with self.connect() as db:
            db.execute("INSERT INTO history(domain,source_text,result_text,created_at) VALUES(?,?,?,?)",
                       (result.domain, result.source_text, result.content, self.now()))

    def history(self, limit: int = 300) -> list[dict]:
        with self.connect() as db:
            return [dict(r) for r in db.execute("SELECT * FROM history ORDER BY id DESC LIMIT ?", (limit,))]

    def get_quick_cache(self, term: str, max_age_days: int = 30) -> dict | None:
        with self.connect() as db:
            row = db.execute(
                "SELECT translation, definition, phonetics, updated_at FROM quick_lookup_cache WHERE term=?",
                (term.strip(),),
            ).fetchone()
        if row is None:
            return None
        try:
            updated = datetime.fromisoformat(row["updated_at"])
            age = datetime.now(timezone.utc) - updated
            if age.days > max_age_days:
                return None
        except (TypeError, ValueError):
            return None
        return dict(row)

    def set_quick_cache(self, term: str, translation: str, definition: str, phonetics: str = "") -> None:
        with self.connect() as db:
            db.execute(
                """INSERT INTO quick_lookup_cache(term,translation,definition,phonetics,updated_at)
                   VALUES(?,?,?,?,?) ON CONFLICT(term) DO UPDATE SET
                   translation=excluded.translation, definition=excluded.definition,
                   phonetics=excluded.phonetics, updated_at=excluded.updated_at""",
                (term.strip(), translation, definition, phonetics, self.now()),
            )

    def export_anki(self, destination: str | Path) -> Path:
        destination = Path(destination)
        destination.parent.mkdir(parents=True, exist_ok=True)
        with destination.open("w", encoding="utf-8-sig", newline="") as stream:
            writer = csv.writer(stream, delimiter=";", quoting=csv.QUOTE_MINIMAL)
            writer.writerow(["Front", "Back", "Example"])
            for word in reversed(self.words()):
                writer.writerow([word["term"], word["meaning"], word["example"]])
        return destination


class EnglishAssistantEngine:
    def __init__(self, store: VocabularyStore | None = None):
        self.store = store or VocabularyStore()
        self.domains = {
            "translation": TranslationDomain(), "dictionary": DictionaryDomain(),
            "grammar": GrammarDomain(), "expressions": ExpressionsDomain(),
            "cefr": CefrDomain(),
        }

    def run(self, domain: str, text: str) -> DomainResult:
        if domain not in self.domains:
            raise DomainError(f"Неизвестный домен: {domain}")
        result = self.domains[domain].run(text)
        self.store.add_history(result)
        return result

    def run_all(self, text: str) -> list[DomainResult]:
        """Run every domain independently; one unavailable online service won't stop others."""
        results: list[DomainResult] = []
        for name, domain in self.domains.items():
            try:
                result = domain.run(text)
            except DomainError as error:
                result = DomainResult(
                    name,
                    f"{name.capitalize()} — ошибка",
                    text,
                    str(error),
                    {"error": True},
                )
            self.store.add_history(result)
            results.append(result)
        return results

from __future__ import annotations

from concurrent.futures import ThreadPoolExecutor, wait
from dataclasses import dataclass

from english_assistant.domains.core import (
    DictionaryDomain,
    DomainError,
    TranslationDomain,
    VocabularyStore,
)
from english_assistant.ocr.word_locator import LocatedWord
from english_assistant.dictionary import get_default_dictionary


@dataclass(frozen=True, slots=True)
class QuickLookupResult:
    word: str
    context: str
    translation: str
    definition: str
    phonetics: str
    confidence: float
    cached: bool = False
    warnings: tuple[str, ...] = ()


class QuickLookupService:
    """Fast path: dictionary + one-word translation only; no full domain fan-out."""

    def __init__(self, store: VocabularyStore | None = None, local_dictionary=None):
        self.store = store or VocabularyStore()
        self.dictionary = DictionaryDomain(timeout=3.5)
        self.translation = TranslationDomain(timeout=3.5)
        self.local_dictionary = local_dictionary if local_dictionary is not None else self._load_local_dictionary()

    @staticmethod
    def _load_local_dictionary():
        return get_default_dictionary()

    def lookup(self, located: LocatedWord) -> QuickLookupResult:
        if self.local_dictionary is not None:
            local_entry = self.local_dictionary.lookup(located.word)
            if local_entry is not None:
                return QuickLookupResult(
                    word=located.word,
                    context=located.context,
                    translation=local_entry.article,
                    definition=f"Офлайн EN–RU · словарная форма: {local_entry.headword}",
                    phonetics="",
                    confidence=located.confidence,
                    cached=False,
                )
        cached = self.store.get_quick_cache(located.word)
        if cached is not None and cached["translation"] and cached["definition"]:
            return QuickLookupResult(
                word=located.word,
                context=located.context,
                translation=cached["translation"],
                definition=cached["definition"],
                phonetics=cached["phonetics"],
                confidence=located.confidence,
                cached=True,
            )

        pool = ThreadPoolExecutor(max_workers=2, thread_name_prefix="quick-lookup")
        tasks = {
            "dictionary": pool.submit(self.dictionary.run, located.word),
            "translation": pool.submit(self.translation.run, located.word),
        }
        warnings = []
        definition = phonetics = translation = ""
        done, pending = wait(tasks.values(), timeout=4.0)
        for name, future in tasks.items():
            if future not in done:
                warnings.append(f"{name.capitalize()}: превышено время ожидания (4 с)")
                continue
            try:
                result = future.result()
                if name == "dictionary":
                    definition = result.content
                    phonetics = " · ".join(result.metadata.get("phonetics", []))
                else:
                    translation = result.content
            except Exception as error:  # external providers are independent
                warnings.append(f"{name.capitalize()}: {error}")
        # Do not block the UI worker while slow providers finish. Network calls
        # have their own bounded timeout and can complete in the background.
        pool.shutdown(wait=False, cancel_futures=True)

        if not definition and not translation:
            raise DomainError("Не удалось получить данные. Проверь интернет или повтори поиск. " + " / ".join(warnings))

        self.store.set_quick_cache(located.word, translation, definition, phonetics)
        return QuickLookupResult(
            word=located.word,
            context=located.context,
            translation=translation or "Перевод недоступен",
            definition=definition or "Словарная статья недоступна",
            phonetics=phonetics,
            confidence=located.confidence,
            warnings=tuple(warnings),
        )

from __future__ import annotations

from PySide6.QtCore import QThread, Signal

from english_assistant.capture.cursor_capture import capture_around_cursor
from english_assistant.lookup.quick_lookup import QuickLookupResult, QuickLookupService
from english_assistant.ocr.word_locator import locate_word
from english_assistant.ocr.uia_text import locate_accessible_word
from english_assistant.ocr.word_locator import LocatedWord


class QuickLookupWorker(QThread):
    succeeded = Signal(object)
    failed = Signal(str)

    def __init__(self, x: int, y: int, parent=None):
        super().__init__(parent)
        self.x = x
        self.y = y

    def run(self):
        try:
            accessible = locate_accessible_word(self.x, self.y)
            if accessible is not None:
                located = LocatedWord(
                    word=accessible.word,
                    context=accessible.context,
                    confidence=1.0,
                    bounds=accessible.bounds or (self.x, self.y, self.x, self.y),
                    method=accessible.method,
                )
            else:
                capture = capture_around_cursor(self.x, self.y)
                located = locate_word(capture)
            result: QuickLookupResult = QuickLookupService().lookup(located)
            self.succeeded.emit(result)
        except Exception as error:  # noqa: BLE001 - worker boundary
            self.failed.emit(str(error))

from __future__ import annotations

from io import BytesIO

from PIL import Image
from PySide6.QtCore import QBuffer, QIODevice
from PySide6.QtGui import QPixmap

from english_assistant.ocr.runtime import _OCR_LOCK, get_engine


def normalize_ocr_lines(lines) -> str:
    if not lines:
        return ""
    return "\n".join(line.strip() for line in lines if line and line.strip())


def pixmap_to_pil(pixmap: QPixmap) -> Image.Image:
    if pixmap.isNull():
        raise ValueError("Получено пустое изображение.")
    buffer = QBuffer()
    buffer.open(QIODevice.OpenModeFlag.WriteOnly)
    if not pixmap.save(buffer, "PNG"):
        raise RuntimeError("Не удалось подготовить изображение для OCR.")
    raw = bytes(buffer.data())
    buffer.close()
    return Image.open(BytesIO(raw)).convert("RGB")


def recognize_text(pixmap: QPixmap) -> str:
    with _OCR_LOCK:
        result = get_engine()(pixmap_to_pil(pixmap))
    lines = getattr(result, "txts", None)
    if lines is None and isinstance(result, tuple) and result:
        lines = result[0]
    return normalize_ocr_lines(lines)

from __future__ import annotations

import re
from dataclasses import dataclass


_WORD = re.compile(r"^[A-Za-z]+(?:['’-][A-Za-z]+)*(?:-[A-Za-z]+)*$")


@dataclass(frozen=True, slots=True)
class AccessibleWord:
    word: str
    context: str
    bounds: tuple[int, int, int, int] | None = None
    method: str = "windows-uia-textpattern"


def locate_accessible_word(x: int, y: int) -> AccessibleWord | None:
    """Get the word at a screen point through Windows UI Automation TextPattern.

    Returns None when the target does not expose TextPattern (common for games,
    canvas-rendered text, video and protected surfaces). OCR remains the fallback.
    """
    try:
        from pywinauto import Desktop

        element = Desktop(backend="uia").from_point(x, y)
        text_pattern = element.iface_text
        word_range = text_pattern.RangeFromPoint((int(x), int(y)))
        word_range.ExpandToEnclosingUnit(2)  # TextUnit.Word
        word = (word_range.GetText(-1) or "").strip()
        if not _WORD.fullmatch(word):
            return None

        context_range = word_range.Clone()
        context_range.ExpandToEnclosingUnit(4)  # TextUnit.Paragraph
        context = (context_range.GetText(500) or word).strip()

        bounds = None
        rectangles = word_range.GetBoundingRectangles()
        if rectangles and len(rectangles) >= 4:
            left, top, width, height = (float(value) for value in rectangles[:4])
            bounds = (round(left), round(top), round(left + width), round(top + height))
        return AccessibleWord(word=word, context=context, bounds=bounds)
    except Exception:
        # UIA providers are inconsistent; never make lookup fail solely because
        # the target application does not expose accessible text.
        return None

from __future__ import annotations

import re
from dataclasses import dataclass

from PIL import Image

from english_assistant.capture.cursor_capture import CursorCapture
from english_assistant.ocr.runtime import _OCR_LOCK, get_engine


@dataclass(frozen=True, slots=True)
class LocatedWord:
    word: str
    context: str
    confidence: float
    bounds: tuple[int, int, int, int]
    method: str = "ocr-estimated-word-box"


_TOKEN_RE = re.compile(r"[A-Za-z]+(?:['’][A-Za-z]+)*(?:-[A-Za-z]+)*")


def _polygon_bounds(polygon) -> tuple[float, float, float, float]:
    points = list(polygon)
    xs = [float(point[0]) for point in points]
    ys = [float(point[1]) for point in points]
    return min(xs), min(ys), max(xs), max(ys)


def _glyph_weight(char: str) -> float:
    if char in "ilI.,'!:;|":
        return 0.42
    if char in "mwMW@%&":
        return 1.35
    if char.isupper():
        return 1.08
    return 0.86


def _word_boxes(text: str, bounds: tuple[float, float, float, float]):
    tokens = list(_TOKEN_RE.finditer(text))
    if not tokens:
        return []
    x1, y1, x2, y2 = bounds
    total = sum(_glyph_weight(ch) for token in tokens for ch in token.group())
    # Allocate a small share to whitespace; this is an OCR fallback, not exact glyph geometry.
    spaces = max(0, len(tokens) - 1)
    total += spaces * 0.55
    cursor = x1
    output = []
    for token in tokens:
        weight = sum(_glyph_weight(ch) for ch in token.group())
        width = (x2 - x1) * weight / total
        output.append((token.group(), (cursor, y1, cursor + width, y2)))
        cursor += width + ((x2 - x1) * 0.55 / total)
    return output


def _distance_to_rect(x: float, y: float, rect: tuple[float, float, float, float]) -> float:
    left, top, right, bottom = rect
    dx = max(left - x, 0.0, x - right)
    dy = max(top - y, 0.0, y - bottom)
    return (dx * dx + dy * dy) ** 0.5


def _unpack_result(result):
    boxes = getattr(result, "boxes", None)
    texts = getattr(result, "txts", None)
    scores = getattr(result, "scores", None)
    if boxes is None and isinstance(result, (tuple, list)) and len(result) >= 2:
        boxes, texts = result[0], result[1]
        scores = result[2] if len(result) > 2 else None
    return ([] if boxes is None else boxes,
            [] if texts is None else texts,
            [] if scores is None else scores)


def locate_word(capture: CursorCapture) -> LocatedWord:
    with _OCR_LOCK:
        result = get_engine()(capture.image)
    boxes, texts, scores = _unpack_result(result)
    candidates = []
    for index, (polygon, text) in enumerate(zip(boxes, texts)):
        if not isinstance(text, str) or not text.strip():
            continue
        local_bounds = _polygon_bounds(polygon)
        global_bounds = (
            local_bounds[0] + capture.origin_x,
            local_bounds[1] + capture.origin_y,
            local_bounds[2] + capture.origin_x,
            local_bounds[3] + capture.origin_y,
        )
        confidence = float(scores[index]) if index < len(scores) else 0.5
        for token, token_bounds in _word_boxes(text, local_bounds):
            global_token_bounds = (
                token_bounds[0] + capture.origin_x,
                token_bounds[1] + capture.origin_y,
                token_bounds[2] + capture.origin_x,
                token_bounds[3] + capture.origin_y,
            )
            distance = _distance_to_rect(capture.cursor_x, capture.cursor_y, global_token_bounds)
            candidates.append((distance, confidence, token, text.strip(), global_token_bounds))

    if not candidates:
        raise ValueError("OCR не обнаружил английское слово рядом с курсором.")

    candidates.sort(key=lambda item: (item[0], -item[1]))
    distance, confidence, word, context, bounds = candidates[0]
    if distance > 24:
        raise ValueError("Курсор не попал в распознанное слово. Наведи его ближе к центру слова.")
    return LocatedWord(
        word=word,
        context=context,
        confidence=confidence,
        bounds=tuple(round(value) for value in bounds),
    )

from PySide6.QtCore import QThread, Signal
from PySide6.QtGui import QPixmap

from english_assistant.ocr.engine import recognize_text


class OcrWorker(QThread):
    succeeded = Signal(str)
    failed = Signal(str)

    def __init__(self, pixmap: QPixmap, parent=None):
        super().__init__(parent)
        self.pixmap = pixmap

    def run(self):
        try:
            self.succeeded.emit(recognize_text(self.pixmap))
        except Exception as error:  # noqa: BLE001 - сообщаем об ошибке через сигнал потока
            self.failed.emit(str(error))

    def run_domain(self, domain: str):
        text = self.input.toPlainText().strip()
        if not text:
            self.status.setText("Сначала введите или распознайте текст.")
            return
        if domain == "dictionary":
            selected = self.input.textCursor().selectedText().strip()
            default_term = selected or text.split()[0].strip(".,!?;:()[]{}\"'")
            term, accepted = QInputDialog.getText(
                self, "Словарь", "Какое слово или выражение разобрать?", text=default_term
            )
            if not accepted or not term.strip():
                return
            text = term.strip()
        self._set_busy(True)
        self.status.setText("Обработка…")
        task = (lambda: self.engine.run_all(text)) if domain == "all" else (lambda: self.engine.run(domain, text))
        worker = TaskWorker(task, self)
        self.workers.append(worker)
        worker.succeeded.connect(self._show_all_results if domain == "all" else self._show_result)
        worker.failed.connect(self._show_error)
        worker.finished.connect(lambda w=worker: self._finished(w))

        for button in self.domain_buttons.values():
            button.setEnabled(not busy)

        blocks = []
        for result in results:
            blocks.append(f"{'=' * 56}\n{result.title}\n{'=' * 56}\n{result.content}")

        self.result_output.setPlainText("\n\n".join(blocks))
        self.tabs.setTabText(0, "Все домены")
        self.tabs.setCurrentIndex(0)
        errors = sum(bool(item.metadata.get("error")) for item in results)
        self.status.setText(f"Обработано доменов: {len(results)} · ошибок: {errors}")
        self.refresh_tables()

            self.workers.remove(worker)
        worker.deleteLater()
        self._set_busy(False)

    def save_word(self):
        source = self.input.textCursor().selectedText().strip() or self.input.toPlainText().strip()
        if not source:
            self.status.setText("Нет текста для сохранения.")
            return
        default_term = source.splitlines()[0].strip()[:120]
        term, accepted = QInputDialog.getText(
            self, "Добавить в словарь", "Слово или выражение:", text=default_term
        )
        if not accepted or not term.strip():
            return
        default_meaning = self.last_result.content if self.last_result else ""
        meaning, accepted = QInputDialog.getMultiLineText(
            self, "Добавить в словарь", "Перевод / значение:", default_meaning
        )
        if not accepted:
            return
        example, accepted = QInputDialog.getMultiLineText(
            self, "Добавить в словарь", "Пример использования:", source
        )
        if not accepted:
            return
        try:
            self.engine.store.add_word(term, meaning, example, source)
        except (ValueError, OSError) as error:
            QMessageBox.warning(self, "Словарь", str(error))
            return
        self.status.setText(f"Сохранено: {term.strip()}")
        self.refresh_tables()
        self.tabs.setCurrentIndex(1)

    def delete_word(self):
        row = self.vocabulary_table.currentRow()
        if row < 0:
            return
        item = self.vocabulary_table.item(row, 0)
        word_id = item.data(Qt.ItemDataRole.UserRole) if item else None
        if word_id is not None:
            self.engine.store.delete_word(int(word_id))
            self.refresh_tables()

    def export_anki(self):
        path, _ = QFileDialog.getSaveFileName(self, "Экспорт для Anki", "english_assistant_anki.csv", "CSV (*.csv)")

        except OSError as error:

            return
        QMessageBox.information(self, "Экспорт Anki", f"Файл сохранён:\n{path}")

    def refresh_tables(self):
        words = self.engine.store.words()
        self.vocabulary_table.setRowCount(len(words))
        for row, word in enumerate(words):
            term_item = QTableWidgetItem(word["term"])
            term_item.setData(Qt.ItemDataRole.UserRole, word["id"])
            self.vocabulary_table.setItem(row, 0, term_item)
            self.vocabulary_table.setItem(row, 1, QTableWidgetItem(word["meaning"]))
            self.vocabulary_table.setItem(row, 2, QTableWidgetItem(word["example"]))
        history = self.engine.store.history()
        self.history_table.setRowCount(len(history))
        for row, item in enumerate(history):
            values = [item["created_at"].replace("T", " ")[:19], item["domain"],
                      item["source_text"], item["result_text"]]
            for column, value in enumerate(values):
                self.history_table.setItem(row, column, QTableWidgetItem(str(value)))

from collections.abc import Callable
from PySide6.QtCore import QThread, Signal


class TaskWorker(QThread):
    succeeded = Signal(object)
    failed = Signal(str)

    def __init__(self, task: Callable, parent=None):
        super().__init__(parent)
        self.task = task

    def run(self):
        try:
            self.succeeded.emit(self.task())
        except Exception as error:  # noqa: BLE001 - передача ошибки в GUI
            self.failed.emit(str(error))
